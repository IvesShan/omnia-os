### 通过型号代码识别

```python
def identify_device(model_code):
    """识别设备型号"""
    device_map = {
        # Mini系列
        'wm160': 'Mini SE',
        'wm161': 'Mini 2',
        'wm1615': 'Mini 2 SE',
        'wm163': 'Mini 3',
        'wm1605': 'Mini 3 Pro',
        'wm170': 'Mini 4 Pro',
        
        # Air系列
        'wm231': 'Air 2S',
        'wm232': 'Mavic Air 2',
        
        # Mavic系列
        'wm240': 'Mavic 3',
        'wm245': 'Mavic 3 Classic',
        'wm246': 'Mavic 3 Pro',
        'wm260': 'Mavic 2 Pro',
        
        # 遥控器
        'rc221': 'RC-N1',
        'rc430': 'RC Pro',
        'rc600': 'RC Plus',
        
        # 眼镜
        'wa140': 'Goggles 2',
        'wa152': 'Goggles 3',
    }
    return device_map.get(model_code, 'Unknown Device')
```

### 通过设备类型编码识别

```python
def identify_module(device_type):
    """识别模块类型"""
    module_map = {
        0x12: 'Perception (感知模块)',
        0x0a: 'Flight Controller (飞控)',
        0x03: 'Camera (相机)',
        0x08: 'Gimbal (云台)',
        0x10: 'PC (电脑端)',
    }
    return module_map.get(device_type, 'Unknown Module')
```
